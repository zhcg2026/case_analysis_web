-- 检查并添加 map 列
SET @dbname = DATABASE();
SET @tablename = 'permissions';
SET @columnname = 'map';
SET @preparedStatement = (SELECT IF(
  (
    SELECT COUNT(*) FROM INFORMATION_SCHEMA.COLUMNS
    WHERE
      (table_schema = @dbname)
      AND (table_name = @tablename)
      AND (column_name = @columnname)
  ) > 0,
  'SELECT 1',
  CONCAT('ALTER TABLE ', @tablename, ' ADD COLUMN ', @columnname, ' INT NOT NULL DEFAULT 0 AFTER chengguantong')
));
PREPARE alterIfNotExists FROM @preparedStatement;
EXECUTE alterIfNotExists;
DEALLOCATE PREPARE alterIfNotExists;

-- 检查并添加 huiwentai 列
SET @columnname = 'huiwentai';
SET @preparedStatement = (SELECT IF(
  (
    SELECT COUNT(*) FROM INFORMATION_SCHEMA.COLUMNS
    WHERE
      (table_schema = @dbname)
      AND (table_name = @tablename)
      AND (column_name = @columnname)
  ) > 0,
  'SELECT 1',
  CONCAT('ALTER TABLE ', @tablename, ' ADD COLUMN ', @columnname, ' INT NOT NULL DEFAULT 0 AFTER map')
));
PREPARE alterIfNotExists FROM @preparedStatement;
EXECUTE alterIfNotExists;
DEALLOCATE PREPARE alterIfNotExists;

-- 查看结果
DESCRIBE permissions;
EOF

### 5. 常见问题和解决方案
- Node.js 版本不兼容 ：
  
  - 解决方案：修改 package.json ，使用兼容 Node.js 18 的 Vite 版本（如 vite@^4.5.3 ）
- 前端构建卡住 ：
  
  - 解决方案：使用 docker-compose 直接构建，避免服务器资源限制
- 数据库连接失败 ：
  
  - 解决方案：确保 DB_HOST 配置为 mysql-case-analysis ，而不是 localhost
- 依赖缺失 ：
  
  - 解决方案：安装必要的 Python 依赖： pip3 install pymysql SQLAlchemy pandas python-docx requests Flask Flask-Cors
