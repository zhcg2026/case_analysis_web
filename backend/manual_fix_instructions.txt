        else:

            }

                except requests.exceptions.Timeout as e:

                        print(f"API 调用超时，{retry_delay}秒后重试... (尝试 {attempt+1}/{max_retries})")
                        import time
                        time.sleep(retry_delay)
                        retry_delay *= 2
                    else:
                        analysis_report = f"API 调用失败：多次尝试后仍然超时 - {str(e)}"
